-DCONFIG_ARDUINO_LOOP_STACK_SIZE=32768

# NimBLE scanner-only mode — disables connection infrastructure entirely.
#
# NIMBLE_BLE_CONNECT = ROLE_CENTRAL || ROLE_PERIPHERAL  (nimble_opt_auto.h).
# With the default (both roles enabled), ble_hs_hci_evt_le_conn_complete() is
# registered in the HCI event dispatch table and ble_hs_conn_init() allocates a
# connection pool at startup.  A BLE device in the environment (e.g. a nearby phone)
# sends a Connection Complete event; NimBLE processes it through that handler while
# the connection pool is in an unexpected state, corrupting the os_mempool and
# crashing with LoadProhibited inside os_mempool_init() (crash20, crash21).
# Setting NIMBLE_BLE_CONNECT=0 removes the handler from the dispatch table and skips
# ble_hs_conn_init() entirely — controller connection events are silently dropped.
-DCONFIG_BT_NIMBLE_ROLE_CENTRAL_DISABLED
-DCONFIG_BT_NIMBLE_ROLE_PERIPHERAL_DISABLED
# We never advertise either; disabling broadcaster saves ~5 KB of flash.
-DCONFIG_BT_NIMBLE_ROLE_BROADCASTER_DISABLED
